! =====================================================================
! SW1  (Cisco Catalyst 2960-24TT) - HQ access switch
! Lab config for Packet Tracer. Passwords are placeholders.
! Connected ports: Fa0/1-2 staff PCs, Fa0/11 guest laptop,
!                  Fa0/21 SRV1, G0/1 trunk to R1-HQ G0/0
! =====================================================================
enable
configure terminal
!
hostname SW1
no ip domain-lookup
enable secret <ENABLE-SECRET>
service password-encryption
banner motd # Authorized access only. Lab device. #
!
! --- VLANs ---
vlan 10
 name STAFF
vlan 20
 name GUEST
vlan 30
 name SERVERS
vlan 99
 name MGMT
vlan 999
 name NATIVE_UNUSED
!
! --- Access ports ---
interface range FastEthernet0/1 - 10
 description STAFF access
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
!
interface range FastEthernet0/11 - 20
 description GUEST access
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
!
interface range FastEthernet0/21 - 23
 description SERVERS access
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
!
! --- Trunk to R1-HQ ---
interface GigabitEthernet0/1
 description Trunk to R1-HQ G0/0
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99,999
 switchport nonegotiate
!
! --- Unused ports: park in VLAN 999 and shut down ---
interface range FastEthernet0/24, GigabitEthernet0/2
 switchport mode access
 switchport access vlan 999
 shutdown
!
! Shut access ports with nothing plugged in (they keep their VLAN;
! use "no shutdown" when a device is added)
interface range FastEthernet0/3 - 10, FastEthernet0/12 - 20, FastEthernet0/22 - 23
 shutdown
!
! --- Management ---
interface Vlan1
 shutdown
interface Vlan99
 description Management
 ip address 10.10.99.2 255.255.255.0
 no shutdown
ip default-gateway 10.10.99.1
!
ip domain-name smallbiz.lab
username admin privilege 15 secret <ADMIN-PASSWORD>
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
!
access-list 10 permit 10.10.10.0 0.0.0.255
!
line console 0
 login local
 exec-timeout 10 0
line vty 0 15
 login local
 transport input ssh
 access-class 10 in
 exec-timeout 10 0
!
end
copy running-config startup-config
